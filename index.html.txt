    <script>
        const apiKey = ""; // Runtime provides the key

        // --- SIMULATOR LOGIC ---
        const range = document.getElementById('dieselRange');
        const display = document.getElementById('dieselValueDisplay');
        const metanolRes = document.getElementById('metanolRes');
        const co2Res = document.getElementById('co2Res');

        function updateSim() {
            const val = parseInt(range.value);
            display.textContent = val.toLocaleString('pt-BR') + ' m³';
            const methanolTons = Math.round(val * 0.85 * 2.1);
            const co2Saved = Math.round(val * 2.6 * 0.95);
            metanolRes.textContent = methanolTons.toLocaleString('pt-BR');
            co2Res.textContent = co2Saved.toLocaleString('pt-BR');
        }

        range.addEventListener('input', updateSim);
        updateSim();

        // --- GEMINI API INTEGRATION ---
        async function callGemini(prompt, systemPrompt = "") {
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemPrompt }] }
            };

            let delay = 1000;
            for (let i = 0; i < 5; i++) {
                try {
                    const response = await fetch(url, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (!response.ok) throw new Error('API request failed');
                    const data = await response.json();
                    return data.candidates?.[0]?.content?.parts?.[0]?.text || "Desculpe, não consegui processar isso agora.";
                } catch (error) {
                    if (i === 4) throw error;
                    await new Promise(r => setTimeout(r, delay));
                    delay *= 2;
                }
            }
        }

        // Feature 1: Intelligence Analysis

        chatToggle.onclick = () => {
            chatWindow.classList.toggle('hidden');
            setTimeout(() => {
                chatWindow.classList.toggle('opacity-0');
                chatWindow.classList.toggle('translate-y-4');
            }, 10);
        };

            chatWindow.classList.add('opacity-0', 'translate-y-4');
            setTimeout(() => chatWindow.classList.add('hidden'), 300);
        };

        async function handleChat() {
            const msg = chatInput.value.trim();
            if (!msg) return;

            // Add user message
            appendMessage(msg, 'user');
            chatInput.value = '';

            // Add loading state
            const loadingId = 'msg-' + Date.now();
            appendMessage('Digitando...', 'ai', loadingId);

            try {
                const sysPrompt = "Você é o Assistente Virtual da GreenMethanol Solutions. Especialista em energia renovável, captura de carbono e e-metanol. Responda de forma clara e educativa em português. Use emojis moderadamente.";
                const response = await callGemini(msg, sysPrompt);
                updateMessage(loadingId, response);
            } catch (e) {
                updateMessage(loadingId, "Erro de conexão. Por favor, tente novamente.");
            }
        }

        function appendMessage(text, side, id = null) {
            const div = document.createElement('div');
            div.id = id;
            div.className = side === 'user' 
                ? 'bg-green-primary text-white p-3 rounded-2xl rounded-tr-none ml-8 shadow-sm'
                : 'bg-white p-3 rounded-2xl rounded-tl-none border border-gray-100 shadow-sm text-gray-700 mr-8';
            div.textContent = text;
            chatMessages.appendChild(div);
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        function updateMessage(id, text) {
            const div = document.getElementById(id);
            if (div) div.textContent = text;
        }
